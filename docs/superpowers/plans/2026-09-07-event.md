## 1.36.2에서 확인한 사실

- 한 턴에서 생기는 이벤트는 모두 같은 `invocation_id`를 가진다.
- 도구 호출 시 function_call 이벤트와 function_response 이벤트의 `author`는 둘 다 에이전트 이름이지만, function_response 이벤트의 `content.role`은 `"user"`다.
- `is_final_response()`는 function_call이나 function_response가 있으면 False, 텍스트만 있으면 True다.
- LlmAgent에 `sub_agents`를 주면 `transfer_to_agent` 도구가 자동으로 붙는다. 모델이 이 도구를 부르면 function_response 이벤트의 `actions.transfer_to_agent`에 대상 이름이 담기고, 이어서 대상 에이전트가 같은 턴 안에서 실행된다.
- BaseAgent 상속 클래스는 `_run_async_impl(self, ctx)`에서 `Event(author=self.name, invocation_id=ctx.invocation_id, ...)`를 yield하면 된다. `actions=EventActions(state_delta={...})`를 실은 이벤트를 yield한 뒤에는 `ctx.session.state`에서 그 값을 읽을 수 있다.

```python
from adk_study.testing import FakeLlm, call_reply, run_turn, text_reply
from agents.event_03_transfer.agent import counter, root_agent


def transfer_to_counter() -> None:
    """부모는 전환을 요청하고 자식은 도구 없이 바로 답하도록 모델을 꾸민다."""

async def test_parent_gets_transfer_tool_automatically():
    transfer_to_counter()
```

```python
"""event_04_custom_event: LLM 없이 Event 를 직접 만들어 yield 한다."""

from collections.abc import AsyncGenerator
from typing import override

from google.adk.agents import BaseAgent
from google.adk.agents.invocation_context import InvocationContext
from google.adk.events import Event, EventActions
from google.genai import types


def _text(text: str) -> types.Content:
    return types.Content(role="model", parts=[types.Part.from_text(text=text)])


class Announcer(BaseAgent):
    """알림 두 개를 이벤트로 내보내는 에이전트."""

    @override
    async def _run_async_impl(
        self, ctx: InvocationContext
    ) -> AsyncGenerator[Event]:
        yield Event(
            author=self.name,
            invocation_id=ctx.invocation_id,
            content=_text("첫 번째 알림"),
        )
        yield Event(
            author=self.name,
            invocation_id=ctx.invocation_id,
            content=_text("두 번째 알림"),
            actions=EventActions(state_delta={"announced": 2}),
        )


root_agent = Announcer(name="event_custom")
```
