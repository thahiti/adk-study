## 1.36.2에서 확인한 사실

- `Runner(app_name=, agent=, session_service=, artifact_service=None, memory_service=None)` 에서 session_service 는 필수다. `InMemoryRunner(agent=, app_name=)` 는 세 서비스를 InMemory 로 채우는 축약이다.
- `runner.run_async(user_id=, session_id=, new_message=, state_delta=None, run_config=None)` 는 Event 를 yield 하는 async generator 다. 사용자 메시지 이벤트는 세션에 저장만 하고 yield 하지 않는다.
- `state_delta=` 를 주면 사용자 메시지 이벤트의 `actions.state_delta` 에 실려 에이전트가 돌기 전에 세션 state 에 반영된다. instruction 의 `{key}` 치환에 바로 쓰인다.
- `RunConfig(max_llm_calls=3)` 을 넘기면 한 턴에서 모델 호출이 3을 넘는 순간 `google.adk.agents.invocation_context.LlmCallsLimitExceededError` 가 난다. 기본값은 500 이다.
- `runner.run(...)` 은 동기 래퍼로, 별도 스레드의 이벤트 루프에서 run_async 를 돌리고 큐로 이벤트를 넘긴다.
- `runner.run_debug("메시지")` 는 세션을 알아서 만들고 대화를 출력한 뒤 `list[Event]` 를 돌려준다.
- `SqliteSessionService("sqlite:///경로")` 는 새 인스턴스로도 같은 파일의 세션을 읽는다. `create_session(session_id=...)` 으로 id 를 지정할 수 있다.
- `uv run python -m agents.<단계>.main` 이 동작한다. agents 가 패키지로 설치되어 있다.

```python
"""runner_01_minimal: Runner 를 직접 만들어 for 루프에서 이벤트를 받는다.

adk web 이 대신 해 주던 일을 스크립트로 옮긴다. 세션 서비스를 만들고,
Runner 에 에이전트와 서비스를 넣고, 세션을 만든 뒤, run_async 가
yield 하는 이벤트를 하나씩 받는다.

실행: uv run python -m agents.runner_01_minimal.main [메시지]
"""

import asyncio
import sys

from google.adk.agents import BaseAgent
from google.adk.events import Event
from google.adk.runners import Runner
from google.adk.sessions import InMemorySessionService
from google.genai import types

from .agent import root_agent

APP_NAME = "runner_01_minimal"
USER_ID = "user"


def describe(event: Event) -> str:
    """이벤트를 한 줄로 요약한다. author 와 종류, 내용 순서다."""
    if event.get_function_calls():
        call = event.get_function_calls()[0]
        return f"[{event.author}] function_call {call.name} {call.args}"
    if event.get_function_responses():
        response = event.get_function_responses()[0].response
        return f"[{event.author}] function_response {response}"
    text = event.content.parts[0].text if event.content else ""
    return f"[{event.author}] text {text}"


async def run(agent: BaseAgent, text: str) -> list[Event]:
    """세션 하나를 만들고 메시지 한 개를 보내 이벤트를 출력하고 모은다."""
    session_service = InMemorySessionService()
    runner = Runner(
        app_name=APP_NAME, agent=agent, session_service=session_service
    )
    session = await session_service.create_session(
        app_name=APP_NAME, user_id=USER_ID
    )
    message = types.Content(
        role="user", parts=[types.Part.from_text(text=text)]
    )
    events: list[Event] = []
    async for event in runner.run_async(
        user_id=USER_ID, session_id=session.id, new_message=message
    ):
        print(describe(event))
        events.append(event)
    return events


if __name__ == "__main__":
    asyncio.run(
        run(root_agent, " ".join(sys.argv[1:]) or "안녕 하세요 글자 수 세 줘")
    )
```

```python
async def run(
    agent: BaseAgent,
    text: str,
    *,
    db_path: str = "sessions.db",
    session_id: str = "runner-demo",
) -> list[Event]:
    """SQLite 세션에 메시지 한 개를 보내 이벤트를 출력하고 모은다.

    같은 db_path 와 session_id 로 다시 부르면 이전 대화에 이어진다.
    """
    session_service = SqliteSessionService(f"sqlite:///{db_path}")
    runner = Runner(
        app_name=APP_NAME,
        agent=agent,
        session_service=session_service,
        artifact_service=InMemoryArtifactService(),
        memory_service=InMemoryMemoryService(),
    )
    session = await session_service.get_session(
        app_name=APP_NAME, user_id=USER_ID, session_id=session_id
    )
    if session is None:
        session = await session_service.create_session(
            app_name=APP_NAME, user_id=USER_ID, session_id=session_id
        )
    message = types.Content(
        role="user", parts=[types.Part.from_text(text=text)]
    )
    events: list[Event] = []
    async for event in runner.run_async(
        user_id=USER_ID, session_id=session.id, new_message=message
    ):
        print(describe(event))
        events.append(event)
    return events
```
